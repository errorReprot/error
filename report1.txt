//DDA

import javax.swing.*;
import java.awt.*;

public class DDATest extends JPanel {

    public void paintComponent(Graphics g) {
        super.paintComponent(g);
        g.setColor(Color.RED);
        dda(g, 50, 50, 400, 300);
    }

    void dda(Graphics g, int x1, int y1, int x2, int y2) {
        int dx = x2 - x1, dy = y2 - y1;
        int steps = Math.max(Math.abs(dx), Math.abs(dy));

        float xi = (float) dx / steps;
        float yi = (float) dy / steps;
        float x = x1, y = y1;

        for (int i = 0; i <= steps; i++) {
            g.fillRect(Math.round(x), Math.round(y), 2, 2);
            x += xi;
            y += yi;
        }
    }

    public static void main(String[] args) {
        JFrame f = new JFrame();
        f.add(new DDATest());
        f.setSize(500, 400);
        f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        f.setVisible(true);
    }
}

//BresenhamLine

import javax.swing.*;
import java.awt.*;

public class BresenhamLineTest extends JPanel {

    public void paintComponent(Graphics g) {
        super.paintComponent(g);
        g.setColor(Color.RED);
        line(g, 50, 50, 400, 300);
    }

    void line(Graphics g, int x1, int y1, int x2, int y2) {
        int dx = Math.abs(x2 - x1);
        int dy = Math.abs(y2 - y1);
        int sx = x1 < x2 ? 1 : -1;
        int sy = y1 < y2 ? 1 : -1;
        int p = dx - dy;

        while (true) {
            g.fillRect(x1, y1, 2, 2);
            if (x1 == x2 && y1 == y2) break;

            int e = 2 * p;

            if (e > -dy) {
                p -= dy;
                x1 += sx;
            }
            if (e < dx) {
                p += dx;
                y1 += sy;
            }
        }
    }

    public static void main(String[] args) {
        JFrame f = new JFrame();
        f.add(new BresenhamLineTest());
        f.setSize(500, 400);
        f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        f.setVisible(true);
    }
}

//Midpoint Circle

import javax.swing.*;
import java.awt.*;

public class MidpointCircleTest extends JPanel {

    public void paintComponent(Graphics g) {
        super.paintComponent(g);
        g.setColor(Color.RED);
        circle(g, 250, 200, 100);
    }

    void circle(Graphics g, int xc, int yc, int r) {
        int x = 0, y = r, p = 1 - r;

        while (x <= y) {
            g.fillRect(xc+x, yc+y, 2, 2);
            g.fillRect(xc-x, yc+y, 2, 2);
            g.fillRect(xc+x, yc-y, 2, 2);
            g.fillRect(xc-x, yc-y, 2, 2);
            g.fillRect(xc+y, yc+x, 2, 2);
            g.fillRect(xc-y, yc+x, 2, 2);
            g.fillRect(xc+y, yc-x, 2, 2);
            g.fillRect(xc-y, yc-x, 2, 2);

            x++;

            if (p < 0)
                p += 2*x + 1;
            else {
                y--;
                p += 2*(x-y) + 1;
            }
        }
    }

    public static void main(String[] args) {
        JFrame f = new JFrame();
        f.add(new MidpointCircleTest());
        f.setSize(500, 450);
        f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        f.setVisible(true);
    }
}

import javax.swing.*;
import java.awt.*;

public class BresenhamCircleTest extends JPanel {

    public void paintComponent(Graphics g) {
        super.paintComponent(g);
        g.setColor(Color.RED);
        circle(g, 250, 200, 100);
    }

    void circle(Graphics g, int xc, int yc, int r) {
        int x = 0, y = r, p = 3 - 2*r;

        while (x <= y) {
            g.fillRect(xc+x, yc+y, 2, 2);
            g.fillRect(xc-x, yc+y, 2, 2);
            g.fillRect(xc+x, yc-y, 2, 2);
            g.fillRect(xc-x, yc-y, 2, 2);
            g.fillRect(xc+y, yc+x, 2, 2);
            g.fillRect(xc-y, yc+x, 2, 2);
            g.fillRect(xc+y, yc-x, 2, 2);
            g.fillRect(xc-y, yc-x, 2, 2);

            if (p < 0)
                p += 4*x + 6;
            else {
                p += 4*(x-y) + 10;
                y--;
            }
            x++;
        }
    }

    public static void main(String[] args) {
        JFrame f = new JFrame();
        f.add(new BresenhamCircleTest());
        f.setSize(500, 450);
        f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
        f.setVisible(true);
    }
}

import java.awt.*;
import javax.swing.*;

public class QuesionAssessment extends JPanel {

	// DDA line drawing
		void dda(Graphics g, int x0, int y0, int x1, int y1) {
			int dx = x1 - x0;
			int dy = y1 - y0;
			int steps = Math.max(Math.abs(dx), Math.abs(dy));

			float xinc = (float) dx / steps;
			float yinc = (float) dy / steps;
			float x = x0;
			float y = y0;

			for (int i = 0; i <= steps; i++) {
				g.fillRect(Math.round(x), Math.round(y), 2, 2);
				x += xinc;
				y += yinc;
			}
		}
		
		
		//draw a shape by joining its points
		void drawPolygon(Graphics g,int [][]points) {
			for(int i=0;i<points.length-1;i++) {
				dda(g,points[i][0],points[i][1],points[i+1][0],points[i+1][1]);
			}
		}
		
		//scaling
		void scale(int [][]points,double s) {
			for(int i=0;i<points.length;i++) {
				points[i][0]=(int)Math.round(points[i][0]*s);
				points[i][1]=(int)Math.round(points[i][1]*s);
				
			}
		}
		
		//rotation
		
		void rotate(int[][]points,double angle) {
			double rad=Math.toRadians(angle);
			double cos=Math.cos(rad);
			double sin=Math.sin(rad);
			
			for(int i=0;i<points.length;i++) {
				double x=points[i][0];
				double y =points[i][1];
				
				points[i][0]=(int)Math.round(x*cos- y*sin);
				points[i][1]=(int)Math.round(x*sin+y*cos);
				
			}
		}
		
		void translation(int [][]points,int tx,int ty) {
			for(int i=0;i<points.length;i++) {
				points[i][0]+=tx;
				points[i][1]+=ty;
				
			}
		}
		
		void drawHouse(Graphics g, Color c, double angle ,int tx,int ty) {
			int[][] base = { {-100,100}, {100,100}, {100,300}, {-100,300}, {-100,100} };
			int[][] roof = { {-100,100}, {0,0}, {100,100}, {-100,100} };
			int[][] door = { {-30,300}, {30,300}, {30,220}, {-30,220}, {-30,300} };

			scale(base,0.6);
			scale(roof,0.6);
			scale(door,0.6);
			
			rotate(base,angle);
			rotate(roof,angle);
			rotate(door,angle);
			
			translation(base,tx,ty);
			translation(roof,tx,ty);
			translation(door,tx,ty);
			
			g.setColor(c);
			drawPolygon(g,base);
			drawPolygon(g,roof);
			drawPolygon(g,door);
						
			
		}
		
		protected void paintComponent(Graphics g) {
			super.paintComponent(g);
			drawHouse(g, Color.red,     0,   300,  40);   // top
			drawHouse(g, Color.green,   90,  560, 300);   // right
			drawHouse(g, Color.orange,  180, 300, 560);   // bottom
			drawHouse(g, Color.magenta, 270,  40, 300);   // left
			
		}
		public static void main(String[] args) {
			JFrame frame = new JFrame("Four Transformed Houses");
	        frame.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
	        frame.setSize(800, 800);
	        frame.add(new QuesionAssessment());
	        frame.setVisible(true);
		}
		
		
}
